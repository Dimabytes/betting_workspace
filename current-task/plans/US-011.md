```python
def _top1_nw_ratio_over_total(networths: Sequence[int]) -> float:
    """Return top-1 net worth divided by the side's total net worth."""
    if not networths:
        return 0.0
    total = sum(networths)
    if total == 0:
        # ponytail: 0.0 when the side sums to 0; unreachable on 5v5 GRID gold
        return 0.0
    return max(networths) / total


def build_top_player_features_over_total(
    radiant: Sequence[int],
    dire: Sequence[int],
) -> TopPlayerFeatures:
    """Three top-player features; ratios are top1 / team total (LoL train)."""
    return TopPlayerFeatures(
        top1_nw_adv=_top1_networth(radiant) - _top1_networth(dire),
        radiant_top1_nw_ratio=_top1_nw_ratio_over_total(radiant),
        dire_top1_nw_ratio=_top1_nw_ratio_over_total(dire),
    )
```

`top1_nw_adv` is the same max−max in both games.
