```python

    """True when at least one assigned profile talks to the Steam Web API."""
    return any(profile.uses_steam for profile in games)

def kalshi_required(games: tuple[GameProfile, ...]) -> bool:
    """True when assigned Dota can bind current KXDOTA2MAP / KXDOTA2GAME series."""
```
